📋 SQL para crear las tablas

        
Ejecutá esto en el SQL Editor de tu proyecto Supabase:

        
CREATE TABLE ventas (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  fecha date NOT NULL, presupuesto text, factura text, cliente text,
  tipo text CHECK (tipo IN ('Servicio','Producto')),
  descripcion text, monto numeric NOT NULL, iva boolean DEFAULT true,
  moneda text DEFAULT 'GS',
  created_at timestamptz DEFAULT now()
);
CREATE TABLE compras (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  fecha date NOT NULL, presupuesto text, factura text, proveedor text,
  tipo text CHECK (tipo IN ('Servicio','Producto')),
  descripcion text, monto numeric NOT NULL, iva boolean DEFAULT true,
  moneda text DEFAULT 'GS',
  created_at timestamptz DEFAULT now()
);
CREATE TABLE presupuestos (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  fecha date NOT NULL, numero text, cliente text, descripcion text,
  monto numeric, estado text DEFAULT 'Pendiente'
    CHECK (estado IN ('Pendiente','Aprobado','Rechazado')),
  notas text, created_at timestamptz DEFAULT now()
);
CREATE TABLE gastos_fijos (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  concepto text NOT NULL, categoria text, monto_mensual numeric NOT NULL,
  frecuencia text DEFAULT 'Mensual', activo boolean DEFAULT true,
  vigente_desde date, vigente_hasta date,
  created_at timestamptz DEFAULT now()
);
-- RLS
ALTER TABLE ventas ENABLE ROW LEVEL SECURITY;
ALTER TABLE compras ENABLE ROW LEVEL SECURITY;
ALTER TABLE presupuestos ENABLE ROW LEVEL SECURITY;
ALTER TABLE gastos_fijos ENABLE ROW LEVEL SECURITY;
CREATE POLICY "all" ON ventas FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "all" ON compras FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "all" ON presupuestos FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "all" ON gastos_fijos FOR ALL USING (true) WITH CHECK (true);

-- SERVICIOS MENSUALES (alojamiento y soporte recurrente)
CREATE TABLE servicios_mensuales (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  cliente text NOT NULL,
  concepto text NOT NULL DEFAULT 'Alojamiento y soporte mensual',
  monto numeric NOT NULL,
  dia_cobro int NOT NULL DEFAULT 1 CHECK (dia_cobro BETWEEN 1 AND 28),
  fecha_inicio date,          -- NULL = pendiente de inicio
  fecha_fin date,             -- NULL = sin fecha de fin
  activo boolean DEFAULT true,
  notas text,
  created_at timestamptz DEFAULT now()
);
CREATE TABLE servicios_cobros (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  servicio_id uuid NOT NULL REFERENCES servicios_mensuales(id) ON DELETE CASCADE,
  periodo text NOT NULL,      -- 'YYYY-MM'
  fecha_cobro date NOT NULL,
  monto numeric NOT NULL,
  created_at timestamptz DEFAULT now(),
  UNIQUE (servicio_id, periodo)
);
ALTER TABLE servicios_mensuales ENABLE ROW LEVEL SECURITY;
ALTER TABLE servicios_cobros ENABLE ROW LEVEL SECURITY;
CREATE POLICY "all" ON servicios_mensuales FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "all" ON servicios_cobros FOR ALL USING (true) WITH CHECK (true);

-- COBROS PENDIENTES
CREATE TABLE cobros_pendientes (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  fecha date NOT NULL,
  presupuesto_id uuid,
  presupuesto_num text,
  factura text,
  cliente text NOT NULL,
  descripcion text,
  monto numeric NOT NULL,
  estado text DEFAULT 'Pendiente'
    CHECK (estado IN ('Pendiente','Cobrado','Vencido')),
  vencimiento date,
  notas text,
  created_at timestamptz DEFAULT now()
);
ALTER TABLE cobros_pendientes ENABLE ROW LEVEL SECURITY;
CREATE POLICY "all" ON cobros_pendientes FOR ALL USING (true) WITH CHECK (true);

-- TALLER TECNICO (ORDENES DE SERVICIO)
CREATE TABLE ordenes_servicio (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  numero text,
  fecha date NOT NULL,
  estado text DEFAULT 'Recibido'
    CHECK (estado IN ('Recibido','En Reparación','Entregado')),
  cliente_nombre text NOT NULL,
  cliente_telefono text,
  cliente_correo text,
  cliente_documento text,
  tipo_equipo text,
  marca text,
  modelo text,
  numero_serie text,
  password_pin text,
  estado_recepcion text,
  falla_mencionada text,
  accesorios text,
  observaciones text,
  trabajos_realizados text,
  presupuesto_id uuid,
  monto_presupuesto numeric,
  forma_pago text,
  anticipo numeric DEFAULT 0,
  fecha_entrega_estimada date,
  created_at timestamptz DEFAULT now()
);
ALTER TABLE ordenes_servicio ENABLE ROW LEVEL SECURITY;
CREATE POLICY "all" ON ordenes_servicio FOR ALL USING (true) WITH CHECK (true);

-- COTIZACION (compartida entre todos los dispositivos)
CREATE TABLE configuracion (
  id int PRIMARY KEY DEFAULT 1,
  cotizacion_usd numeric NOT NULL DEFAULT 7800,
  updated_at timestamptz DEFAULT now(),
  CONSTRAINT single_row CHECK (id = 1)
);
INSERT INTO configuracion (id, cotizacion_usd) VALUES (1, 7800);
ALTER TABLE configuracion ENABLE ROW LEVEL SECURITY;
CREATE POLICY "all" ON configuracion FOR ALL USING (true) WITH CHECK (true);
